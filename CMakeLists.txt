cmake_minimum_required(VERSION 3.22)
project(JerzyMuffCollection VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(JUCE_DIR "" CACHE PATH "Existing JUCE 7.0.12 checkout")
if(JUCE_DIR)
 add_subdirectory("${JUCE_DIR}" juce)
else()
 include(FetchContent)
 FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG 4f43011b96eb0636104cb3e433894cda98243626)
 FetchContent_MakeAvailable(JUCE)
endif()
# GPL build: suppress the temporary JUCE splash in the editor.
get_target_property(muff_juce_modules juce::juce_gui_basics INTERFACE_JUCE_MODULE_PATH)
set_source_files_properties("${muff_juce_modules}/juce_gui_basics/juce_gui_basics.cpp" PROPERTIES COMPILE_DEFINITIONS JUCE_DISPLAY_SPLASH_SCREEN=0)
juce_add_plugin(JerzyMuff COMPANY_NAME Jerzy PLUGIN_MANUFACTURER_CODE Jrzy PLUGIN_CODE Jmuf FORMATS VST3 PRODUCT_NAME "Jerzy Muff Collection" COPY_PLUGIN_AFTER_BUILD FALSE IS_SYNTH FALSE NEEDS_MIDI_INPUT FALSE NEEDS_MIDI_OUTPUT FALSE)
target_sources(JerzyMuff PRIVATE Source/Plugin.cpp)
target_compile_definitions(JerzyMuff PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
target_link_libraries(JerzyMuff PRIVATE juce::juce_audio_utils juce::juce_dsp PUBLIC juce::juce_recommended_config_flags)
enable_testing()
add_executable(dsp_tests tests/DspTests.cpp)
add_test(NAME dsp_regression COMMAND dsp_tests)
juce_add_console_app(processor_tests PRODUCT_NAME "Processor Tests")
target_sources(processor_tests PRIVATE tests/ProcessorTests.cpp)
target_compile_definitions(processor_tests PRIVATE JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0)
target_link_libraries(processor_tests PRIVATE juce::juce_audio_utils juce::juce_dsp juce::juce_recommended_config_flags)
add_test(NAME processor_state COMMAND processor_tests)
